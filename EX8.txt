# F TEST

# Example 1
A = c(14.1,10.1,14.7,13.7,14.0)
B = c(14.0,14.5,13.7,12.7,14.1)
alpha = 0.1
n1 = length(A)
n2 = length(B)
TV = qf(1-alpha,n1-1,n2-1)
F = var(A)/var(B)
TV
F
if(F <= TV) print("Accept H0") else print("Reject H0")

# Example 2 (Nitrogen sources)
x = c(13.4,10.9,11.2,11.8,14.0,15.3,14.2,12.6,17.0,16.2,16.5,15.7)
y = c(12,11.7,10.7,11.2,14.8,13.9,13.7,16.9,16.0,15.6,16.0)
alpha = 0.05
n1 = length(x)
n2 = length(y)
TV = qf(1-alpha,n1-1,n2-1)
F = var(x)/var(y)
TV
F
if(F <= TV) print("Accept H0") else print("Reject H0")

# CHI-SQUARE GOODNESS OF FIT (Given probabilities)
plants = c(20,10,7,4)
chisq.test(plants,p=c(9/16,3/16,3/16,1/16))

# BINOMIAL GOODNESS OF FIT
n = 5
alpha = 0.05
N = 320
p = 0.5
x = 0:n
obf = c(14,56,110,88,40,12)
exf = dbinom(x,n,p)*N
exf
sum(exf)
chisq = sum((obf-exf)^2/exf)
tv = qchisq(1-alpha,n)
chisq
tv
if(chisq <= tv) print("Accept H0") else print("Reject H0")

# POISSON GOODNESS OF FIT
x = 0:6

lambda = sum(f*x)/sum(f)
expf = dpois(x,lambda)*sum(f)
expf

# Grouping
obf = c(275,72,30,15)
exf = c(242,117,28,5)

chisq = sum((obf-exf)^2/exf)
tv = qchisq(0.95,1)
chisq
tv
if(chisq <= tv) print("Accept H0") else print("Reject H0")
