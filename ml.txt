import numpy as np
import matplotlib.pyplot as plt

# Generate sample data
np.random.seed(0)
X = 2 * np.random.rand(100, 1)
y = 4 + 3 * X + np.random.randn(100, 1)

# Add bias term (column of 1s)
X_b = np.c_[np.ones((100, 1)), X]

# Initialize parameters (theta)
theta = np.random.randn(2, 1)

# Hyperparameters
learning_rate = 0.1
n_iterations = 1000
m = len(X_b)

# Gradient Descent
for iteration in range(n_iterations):
    gradients = (2/m) * X_b.T.dot(X_b.dot(theta) - y)
    theta = theta - learning_rate * gradients

print("Learned parameters (theta):")
print(theta)

# Predictions
X_new = np.array([[0], [2]])
X_new_b = np.c_[np.ones((2, 1)), X_new]
y_predict = X_new_b.dot(theta)

print("Predictions:", y_predict)

# Plot results
plt.scatter(X, y, label="Data")
plt.plot(X_new, y_predict, color='red', label="Prediction")
plt.xlabel("X")
plt.ylabel("y")
plt.title("Linear Regression using Gradient Descent")
plt.legend()
plt.show()

# Function definition
def f(x):
    return x**2


# Function definition
def df(x):
    return 2 * x

# Variable assignment or computation
x = 9
# Variable assignment or computation
lr = 0.1
# Variable assignment or computation
iterations = 20

# Variable assignment or computation
history_x = [x]
# Variable assignment or computation
history_y = [f(x)]

# Loop starts here
for _ in range(iterations):
# Variable assignment or computation
    gradient = df(x)
# Variable assignment or computation
    x = x - lr * gradient
    history_x.append(x)
    history_y.append(f(x))

# Variable assignment or computation
x_range = np.linspace(-10, 10, 100)

# Variable assignment or computation
plt.plot(x_range, f(x_range), color='lightgray')
# Variable assignment or computation
plt.scatter(history_x, history_y, color='red')
# Variable assignment or computation
plt.plot(history_x, history_y, color='blue')

plt.title("Gradient Descent")
plt.xlabel("x")
plt.ylabel("f(x)")
