#######################################
#   Z-TEST (SINGLE PROPORTION)        #
#######################################
n = 640
Sprop = 63/n
Pprop = 0.1726
q = 1 - Pprop
alpha = 0.05

# Calculate Z-statistic
z = (Sprop - Pprop) / sqrt(Pprop * q / n)

# Calculate P-value (two-tailed)
pval = 2 * pnorm(-abs(z))

# Decision logic
if(pval > alpha) {
  print("Accept Null hypothesis")
} else {
  print("Reject Null hypothesis")
}

# Optional: Print the results to see the values
print(paste("Z-statistic:", round(z, 4)))
print(paste("P-value:", round(pval, 4)))

#########################################

P=(n1*p1+n2*p2)/(n1+n2)

z=(p1-p2)/sqrt(P*Q*((1/n1)+(1/n2)))
alpha=0.05

if(z<=zalpha){
  print("Accept Null hypothesis")

  print("Reject Null hypothesis")
}

sigma2 = 4     # Population SD 2

n1 = 500       # Sample size 1
n2 = 400       # Sample size 2

# Z calculation
z = ((xbar1 - xbar2) - (mu1 - mu2)) / sqrt((sigma1^2 / n1) + (sigma2^2 / n2))

# Significance level
alpha = 0.05

# Critical value (two-tailed)
z_critical = qnorm(1 - alpha/2)

# p-value
p_value = 2 * (1 - pnorm(abs(z)))

# Output
cat("Z value:", z, "\n")
cat("Critical Z:", z_critical, "\n")
cat("P-value:", p_value, "\n")

# Decision
if(abs(z) > z_critical){
  print("Reject Null Hypothesis")
} else {
  print("Accept Null Hypothesis")
