###############################
#      Z-TEST (SINGLE MEAN)   #
###############################
xbar=14.6
mu0=15.4
sigma=2.5
n=35

z=(xbar-mu0)/(sigma/sqrt(n))
alpha=0.05
zhalfalpha=qnorm(1-(alpha/2))
pval=2*pnorm(z)

if(pval>alpha){
  print("Accept Null hypothesis")
} else{
  print("Reject Null hypothesis")
}

###################################
#     Z-TEST (TWO MEANS)          #
###################################
xbar=20
ybar=15
sigma=4
n1=500
n2=400

z=(xbar-ybar)/(sigma*sqrt((1/n1)+(1/n2)))
alpha=0.05
zalpha=qnorm(1-(alpha/2))

if(z<=zalpha){
  print("Accept Null hypothesis")
} else{
  print("Reject Null hypothesis")
}


#########################################
#   Z-TEST (TWO PROPORTIONS)            #
#########################################
p1=0.20
p2=0.185
n1=900
n2=1600

P=(n1*p1+n2*p2)/(n1+n2)
Q=1-P

z=(p1-p2)/sqrt(P*Q*((1/n1)+(1/n2)))
alpha=0.05
zalpha=qnorm(1-(alpha/2))

if(z<=zalpha){
  print("Accept Null hypothesis")
} else{
  print("Reject Null hypothesis")
}


###################################
#     T-TEST (TWO SAMPLE)         #
###################################
sample1=c(19,17,15,21,16,18,16,14)
sample2=c(15,14,15,19,15,18,16,20)

t=t.test(sample1,sample2)


###################################
#       PAIRED T-TEST             #
###################################
test1=c(19,17,15,21,16,18,16,14,19,20)
test2=c(15,14,15,19,15,18,16,20,22,19)

t=t.test(test1,test2,paired=TRUE)
alpha=0.05
tv=t$p.value

if(tv > alpha){
  print("Accept Ho")
} else{
  print("Reject Ho")
}
